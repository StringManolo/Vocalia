package io.github.stringmanolo.vocalia

import android.app.Application
import io.github.stringmanolo.vocalia.util.CrashHandler
import io.github.stringmanolo.vocalia.util.InternalDebug

class VocaliaApp : Application() {

    override fun onCreate() {
        super.onCreate()

        // 1. Inicializar el logger a disco lo antes posible
        InternalDebug.init(this)
        InternalDebug.mark("VocaliaApp.onCreate START")

        // 2. Instalar el capturador de excepciones Java (no captura SIGSEGV)
        try {
            CrashHandler.install(this)
            InternalDebug.mark("CrashHandler installed")
        } catch (t: Throwable) {
            InternalDebug.mark("CrashHandler install FAILED: ${t.message}")
        }

        // 3. Volcar logcat del proceso anterior (para capturar mensajes nativos
        //    que sobreviven al crash de Sherpa, Vosk, espeak, etc.)
        dumpPreviousLogcat()
    }

    /**
     * Lee las últimas 800 líneas de logcat y filtra las relevantes para
     * el diagnóstico (sherpa, espeak, vocalia, SIGSEGV, Fatal signal...).
     * Las escribe en INTERNAL_DEBUG_FILE para que el usuario pueda leerlas
     * desde Termux sin necesitar adb.
     *
     * Se ejecuta en un hilo aparte para no bloquear el arranque.
     */
    private fun dumpPreviousLogcat() {
        Thread({
            try {
                val proc = Runtime.getRuntime().exec(arrayOf("logcat", "-d", "-t", "800"))
                val out = proc.inputStream.bufferedReader().readText()
                proc.waitFor()

                val filtered = out.lines()
                    .filter { line ->
                        line.contains("sherpa", ignoreCase = true) ||
                        line.contains("espeak", ignoreCase = true) ||
                        line.contains("vocalia", ignoreCase = true) ||
                        line.contains("SIGSEGV", ignoreCase = true) ||
                        line.contains("SIGABRT", ignoreCase = true) ||
                        line.contains("Fatal signal", ignoreCase = true) ||
                        line.contains("libonnxruntime", ignoreCase = true) ||
                        line.contains("dlopen", ignoreCase = true) ||
                        line.contains("UnsatisfiedLink", ignoreCase = true)
                    }
                    .joinToString("\n")

                if (filtered.isNotBlank()) {
                    InternalDebug.mark("[LOGCAT-ANTERIOR] inicio")
                    // Una sola línea con todo el bloque, para no fragmentar el archivo
                    InternalDebug.mark(filtered)
                    InternalDebug.mark("[LOGCAT-ANTERIOR] fin")
                } else {
                    InternalDebug.mark("[LOGCAT-ANTERIOR] sin líneas relevantes")
                }
            } catch (t: Throwable) {
                InternalDebug.mark("[LOGCAT-ANTERIOR] THREW ${t.javaClass.simpleName}: ${t.message}")
            }
        }, "logcat-dump").start()
    }
}

package io.github.stringmanolo.vocalia

import android.Manifest
import android.annotation.SuppressLint
import android.content.Context
import android.content.Intent
import android.content.pm.PackageManager
import android.net.Uri
import android.os.Build
import android.os.Bundle
import android.os.PowerManager
import android.provider.Settings
import android.widget.Button
import android.widget.TextView
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import androidx.core.app.ActivityCompat
import androidx.core.content.ContextCompat

class MainActivity : AppCompatActivity() {

    companion object {
        private const val REQ_NOTIFICATIONS = 200
        private const val REQ_RECORD_AUDIO = 201
        private const val REQ_OVERLAY = 202
    }

    private lateinit var statusText: TextView

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        statusText = findViewById(R.id.statusText)

        requestIgnoreBatteryOptimizations()
        requestNotificationsIfNeeded()
        requestRecordAudioIfNeeded()
        requestOverlayIfNeeded()

        findViewById<Button>(R.id.startButton).setOnClickListener {
            val intent = Intent(this, VocaliaService::class.java)
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) startForegroundService(intent)
            else startService(intent)
            statusText.postDelayed({ updateStatus() }, 300)
        }
        findViewById<Button>(R.id.stopButton).setOnClickListener {
            stopService(Intent(this, VocaliaService::class.java))
            statusText.postDelayed({ updateStatus() }, 300)
        }
        findViewById<Button>(R.id.showLogsButton).setOnClickListener {
            startActivity(Intent(this, LogActivity::class.java))
        }

        updateStatus()
    }

    override fun onResume() {
        super.onResume()
        updateStatus()
    }

    private fun updateStatus() {
        statusText.text = getString(
            if (VocaliaService.isRunning) R.string.status_running else R.string.status_stopped
        )
    }

    @SuppressLint("BatteryLife")
    private fun requestIgnoreBatteryOptimizations() {
        val pm = getSystemService(Context.POWER_SERVICE) as PowerManager
        if (!pm.isIgnoringBatteryOptimizations(packageName)) {
            startActivity(Intent(Settings.ACTION_REQUEST_IGNORE_BATTERY_OPTIMIZATIONS).apply {
                data = Uri.parse("package:$packageName")
            })
        }
    }

    private fun requestNotificationsIfNeeded() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU &&
            ContextCompat.checkSelfPermission(this, Manifest.permission.POST_NOTIFICATIONS)
            != PackageManager.PERMISSION_GRANTED) {
            ActivityCompat.requestPermissions(this,
                arrayOf(Manifest.permission.POST_NOTIFICATIONS), REQ_NOTIFICATIONS)
        }
    }

    private fun requestRecordAudioIfNeeded() {
        if (ContextCompat.checkSelfPermission(this, Manifest.permission.RECORD_AUDIO)
            != PackageManager.PERMISSION_GRANTED) {
            ActivityCompat.requestPermissions(this,
                arrayOf(Manifest.permission.RECORD_AUDIO), REQ_RECORD_AUDIO)
        }
    }

    private fun requestOverlayIfNeeded() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M && !Settings.canDrawOverlays(this)) {
            Toast.makeText(this,
                "Concede 'Dibujar sobre otras apps' para que Vocalia pueda abrir apps desde segundo plano",
                Toast.LENGTH_LONG).show()
            startActivityForResult(
                Intent(Settings.ACTION_MANAGE_OVERLAY_PERMISSION,
                    Uri.parse("package:$packageName")),
                REQ_OVERLAY
            )
        }
    }
}

class VocaliaService : Service() {

    companion object {
        private const val NOTIFICATION_ID = 1001

        @Volatile var isRunning: Boolean = false
            private set

    private lateinit var ttsManager: TtsManager

    override fun onCreate() {

        InternalDebug.mark("Service: onCreate")

        appLauncher = AppLauncher(this)
        appLauncher.preload()
        val appNames = appLauncher.getAllInstalledAppNames()
        val aliases = appLauncher.getKnownAliases()
        InternalDebug.mark("Service: ${appNames.size} apps, ${aliases.size} aliases")
        val grammar = buildGrammar(appNames, aliases)
        InternalDebug.mark("Service: grammar de ${grammar.length} chars")

        ttsManager = TtsManager(this)
        voskEngine = VoskEngine(this)
        core = VocaliaCore(
            voskEngine = voskEngine,
            ttsManager = ttsManager,
            commandParser = CommandParser(appLauncher),
            appLauncher = appLauncher,
            log = { msg -> InternalDebug.mark("[core] $msg") }
        )

        ttsManager.onEngineReady = {
            InternalDebug.mark("Service: TTS listo, arrancando core")
            core.start(grammar)

        ttsManager.init()

        return builder
            .setContentTitle(getString(R.string.notification_title))
            .setContentText(getString(R.string.notification_text))
            .setSmallIcon(android.R.drawable.ic_btn_speak_now)
            .setContentIntent(contentIntent)
            .setOngoing(true)
            .build()
    }

package io.github.stringmanolo.vocalia

import android.accessibilityservice.AccessibilityService
import android.view.accessibility.AccessibilityEvent
import android.view.accessibility.AccessibilityNodeInfo
import io.github.stringmanolo.vocalia.util.InternalDebug

/**
 * Servicio de accesibilidad de Vocalia.
 *
 * Expone una lista ordenada de nodos "interactivos" (BFS) y métodos para
 * navegar por ella, leer su descripción, pulsarla, hacer scroll, ir atrás
 * y pegar del portapapeles.
 *
 * Toda la operativa se asume llamada desde el hilo principal (VocaliaCore
 * lo hace a través de mainHandler.post).
 */
class VocaliaAccessibilityService : AccessibilityService() {

    companion object {
        private const val MAX_NODES = 120

        @Volatile
        var instance: VocaliaAccessibilityService? = null
            private set
    }

    enum class ScrollDirection { UP, DOWN, LEFT, RIGHT }

    /** Lista actual de nodos interactivos (orden BFS). */
    private var interactiveNodes: List<AccessibilityNodeInfo> = emptyList()

    /** Índice actual sobre `interactiveNodes`. -1 si no hay selección. */
    private var currentIndex: Int = -1

    private fun dbg(msg: String) {
        InternalDebug.mark("[A11y] $msg")
    }

    // ----------------------------------------------------------------
    // Ciclo de vida
    // ----------------------------------------------------------------

    override fun onServiceConnected() {
        super.onServiceConnected()
        instance = this
        dbg("onServiceConnected")
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {
        // Stub. En fases posteriores refrescaremos en TYPE_WINDOW_STATE_CHANGED.
    }

    override fun onInterrupt() {
        dbg("onInterrupt")
    }

    override fun onDestroy() {
        if (instance === this) instance = null
        dbg("onDestroy")
        super.onDestroy()
    }

    // ----------------------------------------------------------------
    // Construcción de la lista de nodos
    // ----------------------------------------------------------------

    /**
     * Recorre el árbol de la ventana activa (BFS) y construye una lista con
     * los nodos visibles que son interactuables o contienen texto.
     */
    fun refreshNodeList() {
        val root = rootInActiveWindow
        if (root == null) {
            interactiveNodes = emptyList()
            currentIndex = -1
            dbg("refreshNodeList: rootInActiveWindow=null")
            return
        }
        val result = mutableListOf<AccessibilityNodeInfo>()
        val queue = ArrayDeque<AccessibilityNodeInfo>()
        queue.add(root)

        while (queue.isNotEmpty() && result.size < MAX_NODES) {
            val node = queue.removeFirst()
            if (!node.isVisibleToUser) continue

            val hasText = !node.text.isNullOrBlank() ||
                          !node.contentDescription.isNullOrBlank()
            val isInteractive = node.isClickable || node.isFocusable || node.isEditable

            if (isInteractive || hasText) {
                result.add(node)
            }
            for (i in 0 until node.childCount) {
                node.getChild(i)?.let { queue.add(it) }
            }
        }

        interactiveNodes = result
        currentIndex = if (result.isNotEmpty()) 0 else -1
        dbg("refreshNodeList: ${result.size} nodos, index=$currentIndex")
    }

    private fun getCurrentNode(): AccessibilityNodeInfo? =
        if (currentIndex in interactiveNodes.indices) interactiveNodes[currentIndex] else null

    /**
     * Genera una descripción natural de un nodo: texto + contentDescription
     * + tipo (botón, campo de texto, casilla…) + pista opcional.
     */
    private fun describeNode(node: AccessibilityNodeInfo): String {
        val text  = node.text?.toString()?.takeIf { it.isNotBlank() }
        val desc  = node.contentDescription?.toString()?.takeIf { it.isNotBlank() }
        val hint  = node.hintText?.toString()?.takeIf { it.isNotBlank() }

        val type = when {
            node.isEditable   -> "campo de texto"
            node.isCheckable  -> "casilla"
            node.isClickable  -> "botón"
            node.isFocusable  -> "elemento"
            else              -> "texto"
        }

        val content = when {
            text != null && desc != null && text != desc -> "$text, $desc"
            text != null -> text
            desc != null -> desc
            else -> ""
        }

        val hintSuffix = if (hint != null) ", pista: $hint" else ""

        return if (content.isBlank()) type
               else "$content. $type$hintSuffix"
    }

    // ----------------------------------------------------------------
    // Navegación
    // ----------------------------------------------------------------

    fun readCurrentNode(): String {
        if (interactiveNodes.isEmpty()) {
            return "no hay elementos. Prueba a decir siguiente o anterior."
        }
        val node = getCurrentNode() ?: return "no hay elemento seleccionado"
        val desc = describeNode(node)
        dbg("read [$currentIndex/${interactiveNodes.size - 1}] -> \"$desc\"")
        return desc
    }

    fun nextNode(): String {
        if (interactiveNodes.isEmpty()) refreshNodeList()
        if (interactiveNodes.isEmpty()) return "no hay elementos"
        currentIndex = if (currentIndex < 0) 0 else (currentIndex + 1) % interactiveNodes.size
        return readCurrentNode()
    }

    fun previousNode(): String {
        if (interactiveNodes.isEmpty()) refreshNodeList()
        if (interactiveNodes.isEmpty()) return "no hay elementos"
        currentIndex = if (currentIndex <= 0) interactiveNodes.size - 1 else currentIndex - 1
        return readCurrentNode()
    }

    // ----------------------------------------------------------------
    // Acciones
    // ----------------------------------------------------------------

    fun clickCurrentNode(): String {
        val node = getCurrentNode() ?: return "no hay elemento seleccionado"
        if (!node.refresh()) {
            refreshNodeList()
            return "el elemento ya no está disponible, lista actualizada"
        }
        if (!node.isClickable) return "este elemento no se puede pulsar"
        val ok = node.performAction(AccessibilityNodeInfo.ACTION_CLICK)
        dbg("click [$currentIndex] -> ok=$ok")
        return if (ok) "pulsado" else "no se pudo pulsar"
    }

    fun performBack(): String {
        val ok = performGlobalAction(GLOBAL_ACTION_BACK)
        dbg("back -> ok=$ok")
        refreshNodeList()
        return if (ok) "atrás" else "no se pudo volver atrás"
    }

    fun scroll(direction: ScrollDirection): String {
        val node = getCurrentNode() ?: rootInActiveWindow
        if (node == null) return "no hay ventana activa"
        val action = when (direction) {
            ScrollDirection.UP, ScrollDirection.LEFT ->
                AccessibilityNodeInfo.ACTION_SCROLL_BACKWARD
            ScrollDirection.DOWN, ScrollDirection.RIGHT ->
                AccessibilityNodeInfo.ACTION_SCROLL_FORWARD
        }
        val ok = node.performAction(action)
        dbg("scroll $direction -> ok=$ok (nodo=${node.className})")
        refreshNodeList()
        return if (ok) "desplazado" else "no se pudo desplazar"
    }

    fun pasteClipboard(): String {
        val root = rootInActiveWindow ?: return "no hay ventana activa"
        val focused = root.findFocus(AccessibilityNodeInfo.FOCUS_INPUT)
            ?: return "no hay campo de texto enfocado"
        val ok = focused.performAction(AccessibilityNodeInfo.ACTION_PASTE)
        dbg("paste -> ok=$ok (foco=${focused.className})")
        return if (ok) "pegado" else "no se pudo pegar"
    }
}

package io.github.stringmanolo.vocalia

import android.content.Context
import android.media.AudioManager
import android.media.ToneGenerator
import android.util.Log
import org.vosk.Model
import org.vosk.Recognizer
import org.vosk.android.RecognitionListener
import org.vosk.android.SpeechService
import org.vosk.android.StorageService
import java.io.IOException

class VoskEngine(private val context: Context) {

    companion object {
        private const val TAG = "VoskEngine"
        private const val DEFAULT_GRAMMAR = """["asistente", "asistentes", "[unk]"]"""
    }

    private var model: Model? = null
    private var speechService: SpeechService? = null
    @Volatile private var currentGrammar: String = DEFAULT_GRAMMAR

    var onResult: ((String) -> Unit)? = null
    var onPartialResult: ((String) -> Unit)? = null
    var onError: ((String) -> Unit)? = null

    private val listener = object : RecognitionListener {
        override fun onPartialResult(hypothesis: String?) { onPartialResult?.invoke(hypothesis ?: "") }
        override fun onResult(hypothesis: String?) { onResult?.invoke(hypothesis ?: "") }
        override fun onFinalResult(hypothesis: String?) {}
        override fun onError(exception: Exception?) {
            Log.e(TAG, "Error reconocimiento", exception)
            onError?.invoke(exception?.message ?: "Error")
        }
        override fun onTimeout() {}
    }

    fun init(grammar: String, onReady: () -> Unit) {
        currentGrammar = grammar
        StorageService.unpack(
            context,
            "vosk-model-small-es-0.42",
            "model",
            { loadedModel ->
                model = loadedModel
                Log.i(TAG, "Modelo cargado, grammar de ${grammar.length} chars")
                onReady()
            },
            { exception ->
                Log.e(TAG, "Error cargando modelo", exception)
                onError?.invoke("Error cargando modelo: ${exception.message}")
            }
        )
    }

    fun startListening() {
        if (speechService != null) return
        val m = model ?: run {
            Log.w(TAG, "startListening sin modelo")
            return
        }
        try {
            val recognizer = Recognizer(m, 16000.0f, currentGrammar)
            speechService = SpeechService(recognizer, 16000.0f)
            speechService?.startListening(listener)
            Log.i(TAG, "startListening OK (grammar=${currentGrammar.length} chars)")
        } catch (e: IOException) {
            Log.e(TAG, "startListening falló", e)
            onError?.invoke("Error startListening: ${e.message}")
        }
    }

    fun stopListening() {
        speechService?.stop()
        speechService = null
    }

    /** Cambia la gramática activa. Si el servicio está escuchando, lo reinicia con la nueva. */
    fun setGrammar(grammar: String) {
        if (grammar == currentGrammar && speechService != null) return
        currentGrammar = grammar
        Log.i(TAG, "setGrammar (${grammar.length} chars), speechService=${speechService != null}")
        if (speechService != null) {
            stopListening()
            startListening()
        }
    }

    fun shutdown() {
        stopListening()
        model?.close()
        model = null
    }

    fun beep() {
        Thread {
            var tg: ToneGenerator? = null
            try {
                tg = ToneGenerator(AudioManager.STREAM_MUSIC, 60)
                tg.startTone(ToneGenerator.TONE_PROP_BEEP, 100)
                Thread.sleep(150)
            } catch (_: Throwable) {} finally {
                try { tg?.release() } catch (_: Throwable) {}
            }
        }.start()
    }
}

class CommandParser(private val appLauncher: AppLauncher? = null) {

    private val esLocale = Locale("es", "ES")

    private val openPrefixes = listOf(
        "abre la app ", "abre el ", "abre la ", "abre ",
        "abrir la app ", "abrir el ", "abrir la ", "abrir "
    )

    private val closePrefixes = listOf(
        "cierra la app ", "cierra el ", "cierra la ", "cierra ",
        "cerrar la app ", "cerrar el ", "cerrar la ", "cerrar ",
        "mata la app ", "mata el ", "mata la ", "mata ",
        "matar la app ", "matar el ", "matar la ", "matar "
    )

    fun parse(raw: String): CommandResult {
        val text = raw.lowercase(esLocale).trim()
        if (text.isBlank()) return CommandResult.Response("no te he entendido")

        // ============================================================
        // Modo usar (UI navigation) — entrada / salida
        // ============================================================
        if (matches(text,
                "modo usar", "activar modo usar", "activa el modo usar",
                "entra al modo usar", "entrar al modo usar",
                "modo navegacion", "modo navegación",
                "activar modo navegacion", "activa el modo navegacion")) {
            return CommandResult.EnterUiMode
        }
        if (matches(text,
                "sal del modo usar", "salir del modo usar", "salir del modo",
                "desactiva el modo usar", "termina el modo usar", "fin del modo usar",
                "sal del modo navegacion", "salir del modo navegacion")) {
            return CommandResult.ExitUiMode
        }

        // ============================================================
        // Comandos dentro del modo usar
        // ============================================================
        if (matches(text,
                "arrastra arriba", "arrastra elemento arriba",
                "scroll arriba", "desplaza arriba", "sube")) {
            return CommandResult.UiScroll(CommandResult.UiDirection.UP)
        }
        if (matches(text,
                "arrastra abajo", "arrastra elemento abajo",
                "scroll abajo", "desplaza abajo", "baja")) {
            return CommandResult.UiScroll(CommandResult.UiDirection.DOWN)
        }
        if (matches(text,
                "arrastra izquierda", "arrastra elemento izquierda",
                "scroll izquierda", "desplaza izquierda")) {
            return CommandResult.UiScroll(CommandResult.UiDirection.LEFT)
        }
        if (matches(text,
                "arrastra derecha", "arrastra elemento derecha",
                "scroll derecha", "desplaza derecha")) {
            return CommandResult.UiScroll(CommandResult.UiDirection.RIGHT)
        }
        if (matches(text, "siguiente", "avanza",
                "siguiente elemento", "el siguiente")) {
            return CommandResult.UiNext
        }
        if (matches(text, "anterior", "retrocede",
                "anterior elemento", "el anterior")) {
            return CommandResult.UiPrevious
        }
        if (matches(text,
                "lee elemento", "leer elemento",
                "lee", "leer", "léelo", "leelo")) {
            return CommandResult.UiRead
        }
        if (matches(text,
                "pulsa", "pulsar", "toca", "tocar",
                "activa elemento", "activar elemento",
                "click", "clic")) {
            return CommandResult.UiClick
        }
        if (matches(text,
                "atrás", "atras", "back",
                "volver atrás", "vuelve atrás", "regresa")) {
            return CommandResult.UiBack
        }
        if (matches(text,
                "portapapeles pegar", "pega el portapapeles",
                "pega del portapapeles", "pega", "pegar")) {
            return CommandResult.UiPaste
        }

        // ============================================================
        // Resto de comandos (ya existentes)
        // ============================================================
        return when {
            matches(text, "ayuda", "qué puedes hacer", "que puedes hacer",
                "qué sabes hacer", "que sabes hacer",
                "dime todo lo que puedes hacer", "dime lo que puedes hacer",
                "guía", "guia", "comandos",
                "lista de comandos", "qué comandos tienes", "que comandos tienes") ->
                CommandResult.Help

            matches(text, "lístame las apps", "listame las apps",
                "lista de apps disponibles", "lista completa de apps",
                "qué apps disponibles", "que apps disponibles",
                "todas las apps", "todas las aplicaciones") ->
                CommandResult.ListAppsVerbose

            matches(text, "lista de apps", "lista apps",
                "qué apps tengo", "que apps tengo", "lista de aplicaciones") ->
                CommandResult.ListApps

            matches(text, "sal de escucha continua", "desactiva escucha continua",
                "desactiva modo continuo") ->
                CommandResult.ExitContinuous

            matches(text, "escucha continua", "modo continuo", "escucha constante") ->
                CommandResult.EnterContinuous

            matches(text, "pausa el tts", "pausa el texto", "pausa la lectura",
                "pausa lectura", "pausa voz",
                "para el tts", "para el texto", "para la lectura",
                "pausa", "para", "parar", "pausar", "silencio", "callate",
                "callá", "stop", "detente", "basta", "alto", "quieto") ->
                CommandResult.PauseTts

            matches(text, "continua el tts", "continua el texto",
                "continua la lectura", "continua lectura", "continua voz",
                "reanuda el tts", "reanuda la lectura", "sigue leyendo",
                "continua", "continúa", "reanuda", "sigue") ->
                CommandResult.ResumeTts

            extractCloseAppName(text) != null ->
                CommandResult.CloseApp(extractCloseAppName(text)!!)

            matches(text, "cierra esta app", "cierra la app actual", "cierra esto",
                "cierra la app que esté abierta", "cierra lo que hay delante") ->
                CommandResult.CloseForegroundApp

            extractAppName(text) != null -> CommandResult.OpenApp(extractAppName(text)!!)
            appLauncher?.hasApp(text) == true -> CommandResult.OpenApp(text)

            matches(text, "descansa", "descanso", "descansar",
                "cansa", "duerme", "dormir") ->
                CommandResult.Sleep

            matches(text, "inteligente", "piper", "neuronal", "natural") ->
                CommandResult.SetVoiceMode(intelligent = true)
            matches(text, "por defecto", "espeak", "robotica", "robotizada") ->
                CommandResult.SetVoiceMode(intelligent = false)

            matches(text, "hora") -> CommandResult.Response(currentTime())
            matches(text, "fecha", "dia") -> CommandResult.Response(currentDate())
            matches(text, "hola") -> CommandResult.Response("hola")
            matches(text, "estas", "tal") -> CommandResult.Response("bien, gracias")

            else -> CommandResult.Response("no entiendo")
        }
    }

    private fun extractAppName(text: String): String? {
        for (p in openPrefixes) {
            if (text.startsWith(p)) {
                val raw = text.removePrefix(p).trim()
                val stripped = raw
                    .removeSuffix(" callate").removeSuffix(" calla")
                    .removeSuffix(" por favor").removeSuffix(" ahora").removeSuffix(" ya")
                    .trim()
                return if (stripped.isNotBlank()) stripped else null
            }
        }
        return null
    }

    private fun extractCloseAppName(text: String): String? {
        for (p in closePrefixes) {
            if (text.startsWith(p)) {
                val raw = text.removePrefix(p).trim()
                val stripped = raw
                    .removeSuffix(" callate").removeSuffix(" calla")
                    .removeSuffix(" por favor").removeSuffix(" ahora").removeSuffix(" ya")
                    .trim()
                return if (stripped.isNotBlank()) stripped else null
            }
        }
        return null
    }

    private fun matches(text: String, vararg keywords: String): Boolean {
        val words = text.split("\\s+".toRegex()).filter { it.isNotBlank() }
        return keywords.any { kw ->
            if (kw.contains(" ")) text.contains(kw)
            else words.any { w -> w == kw || (w.startsWith(kw) && w.length - kw.length <= 3) }
        }
    }

    private fun currentTime(): String {
        val fmt = SimpleDateFormat("H 'y' mm", esLocale)
        return "son las ${fmt.format(Date())}"
    }

    private fun currentDate(): String {
        val fmt = SimpleDateFormat("EEEE d 'de' MMMM", esLocale)
        return "hoy es ${fmt.format(Date())}"
    }
}

        resumeRunnable = Runnable {
            if (state != State.EXECUTING && state != State.TTS_SPEAKING &&
                state != State.TTS_PAUSED && !isBlockingHelp) {
                voskEngine.startListening()
                log("Vosk reanudado tras ${resumeDelayMs}ms (state=$state)")
            }
        }.also { mainHandler.postDelayed(it, resumeDelayMs) }
    }

    // Helpers de TTS + accesibilidad
    // ----------------------------------------------------------------

    private fun speakAndReturn(text: String, after: () -> Unit) {
        ttsManager.speak(text,
            onStart = { enterTtsSpeaking() },
            onComplete = {
                if (state == State.TTS_SPEAKING) {
                    state = State.EXECUTING
                    after()
                }
            })
    }

    private fun speakBlocking(text: String, after: () -> Unit) {
        isBlockingHelp = true
        cancelTimeouts()
        voskEngine.stopListening()
        log("Vosk detenido por TTS bloqueante (Ayuda)")
        state = State.TTS_SPEAKING
        ttsManager.speak(
            text,
            onStart = { log("TTS bloqueante en curso") },
            onComplete = {
                isBlockingHelp = false
                state = State.EXECUTING
                after()
            }
        )
    }

    /**
     * Ejecuta una acción de accesibilidad (que debe correr en el hilo principal)
     * y devuelve el texto que produce al hilo del TTS.
     * Si el servicio no está disponible, sale del modo usar.
     */
    private fun a11yCommand(action: (VocaliaAccessibilityService) -> String) {
        mainHandler.post {
            val svc = VocaliaAccessibilityService.instance
            if (svc == null) {
                log("a11y: servicio no disponible")
                speakAndReturn("servicio de accesibilidad no disponible, actívalo en ajustes") {
                    enterIdle()
                }
                return@post
            }
            val msg = try {
                action(svc)
            } catch (t: Throwable) {
                log("a11y THREW ${t.javaClass.simpleName}: ${t.message}")
                "error de accesibilidad"
            }
            log("a11y result: \"$msg\"")
            speakAndReturn(msg) {
                if (state == State.EXECUTING) enterUiNavigation() else enterIdle()
            }
        }
    }

    // Procesamiento de resultados Vosk
    // ----------------------------------------------------------------

    private fun handleResult(json: String) {
        val text = extractText(json)
        if (text.isBlank() || text == "[unk]") return

        if (isBlockingHelp) {
            log("result: \"$text\" IGNORADO (isBlockingHelp)")
            return
        }

        log("result: \"$text\" (state=$state)")

        when (state) {
            State.IDLE -> {
                val (hasWake, remainder) = extractWake(text)
                if (hasWake) {
                    enterListening()
                    if (remainder.isNotBlank()) handleCommand(remainder, State.IDLE)
                }
            }
            State.LISTENING -> {
                val (hasWake, remainder) = extractWake(text)
                if (hasWake && remainder.isNotBlank()) {
                    cancelTimeouts(); voskEngine.beep(); handleCommand(remainder, State.IDLE)
                } else if (hasWake) {
                    cancelTimeouts(); voskEngine.beep(); enterListening()
                } else {
                    handleCommand(text, State.IDLE)
                }
            }
            State.CONTINUOUS -> {
                val (hasWake, remainder) = extractWake(text)
                val command = if (hasWake && remainder.isNotBlank()) remainder else text
                handleCommand(command, State.CONTINUOUS)
            }
            State.UI_NAVIGATION -> {
                val (hasWake, remainder) = extractWake(text)
                val command = when {
                    hasWake && remainder.isNotBlank() -> remainder
                    hasWake -> ""
                    else -> text
                }
                if (command.isBlank()) return
                handleCommand(command, State.UI_NAVIGATION)
            }
            State.TTS_SPEAKING -> {
                val (hasWake, _) = extractWake(text)
                val textLower = text.lowercase()
                val words = textLower.split("\\s+".toRegex()).filter { it.isNotBlank() }
                val userPauseWord = words.firstOrNull { it in allPauseWords }

                when {
                    userPauseWord == null ->
                        log("TTS_SPEAKING: no pause word, ignorando")
                    hasWake -> {
                        log("TTS_SPEAKING: pausa por '$userPauseWord' (con wake word)")
                        ttsManager.pause()
                        enterTtsPaused()
                    }
                    else ->
                        log("TTS_SPEAKING: '$userPauseWord' sin wake word, ignorando (anti-eco)")
                }
            }
            State.TTS_PAUSED -> {
                val (hasWake, remainder) = extractWake(text)
                val command = when {
                    hasWake && remainder.isNotBlank() -> remainder
                    hasWake -> ""
                    else -> text
                }
                if (command.isBlank()) return
                val parsed = commandParser.parse(command)
                when (parsed) {
                    is CommandResult.ResumeTts -> {
                        log("TTS_PAUSED: reanudar")
                        if (!ttsManager.resumePaused(
                                onStart = { enterTtsSpeaking() },
                                onComplete = {
                                    if (state == State.TTS_SPEAKING) resumeListeningWithDelay()
                                })) {
                            log("TTS_PAUSED: no había nada que reanudar")
                            enterIdle()
                        }
                    }
                    CommandResult.Sleep, is CommandResult.ExitContinuous -> {
                        log("TTS_PAUSED: cancelar")
                        ttsManager.stop()
                        speakAndReturn("descanso") { enterIdle() }
                    }
                    else -> {
                        log("TTS_PAUSED: comando nuevo, se abandona la lectura pausada")
                        ttsManager.clearPausedSpeech()
                        ttsManager.stop()
                        handleCommand(command, State.IDLE)
                    }
                }
            }
            State.EXECUTING -> log("ignorando result (EXECUTING)")
        }
    }

    // ----------------------------------------------------------------
    // Ejecución de comandos
    // ----------------------------------------------------------------

    private fun handleCommand(text: String, returnTo: State) {
        log("handleCommand: \"$text\" returnTo=$returnTo")
        if (ttsManager.isSpeaking()) {
            log("handleCommand: cancelando TTS en curso")
            ttsManager.stop()
        }
        val parsed = commandParser.parse(text)
        val forceIdle = parsed is CommandResult.Sleep
            || parsed is CommandResult.ExitContinuous
            || parsed is CommandResult.ExitUiMode
        enterExecuting(if (forceIdle) State.IDLE else returnTo)

        when (parsed) {
            CommandResult.Sleep ->
                speakAndReturn("descanso") { enterIdle() }
            CommandResult.ExitContinuous ->
                speakAndReturn("escucha continua desactivada") { enterIdle() }
            CommandResult.EnterContinuous ->
                speakAndReturn("escucha continua activada") { enterContinuous() }
            CommandResult.PauseTts -> {
                log("PauseTts fuera de TTS_SPEAKING, ignorado")
                afterExecution()
            }
            CommandResult.ResumeTts -> {
                if (ttsManager.hasPausedSpeech()) {
                    log("ResumeTts con buffer pendiente: reanudando")
                    if (!ttsManager.resumePaused(
                            onStart = { enterTtsSpeaking() },
                            onComplete = { if (state == State.TTS_SPEAKING) resumeListeningWithDelay() })) {
                        log("ResumeTts: resumePaused falló, ignorado")
                        afterExecution()
                    }
                } else {
                    log("ResumeTts fuera de TTS_PAUSED, ignorado")
                    afterExecution()
                }
            }
            CommandResult.Help -> {
                log("Help solicitado (bloqueante, Vosk detenido)")
                speakBlocking(HELP_TEXT) { afterExecution() }
            }
            CommandResult.ListApps -> {
                val all = appLauncher.getAllAppsForTTS()
                log("ListApps: ${all.size} apps")
                val msg = if (all.isEmpty()) "no tengo apps instaladas"
                          else "tengo ${all.size} apps instaladas. Son: ${all.joinToString(", ")}"
                speakAndReturn(msg) { afterExecution() }
            }
            CommandResult.ListAppsVerbose -> {
                val all = appLauncher.getAllAppsForTTS()
                log("ListAppsVerbose: ${all.size} apps")
                all.forEach { app ->
                    val aliases = appLauncher.getAliasesForApp(app)
                    if (aliases.isEmpty()) log("ListAppsVerbose:   - $app")
                    else log("ListAppsVerbose:   - $app (di: ${aliases.joinToString(" / ")})")
                }
                val msg = if (all.isEmpty()) "no tengo apps instaladas"
                          else "tengo ${all.size} apps instaladas. Son: ${all.joinToString(", ")}"
                speakAndReturn(msg) { afterExecution() }
            }
            is CommandResult.Response ->
                speakAndReturn(parsed.text) { afterExecution() }
            is CommandResult.SetVoiceMode -> {
                if (parsed.intelligent) {
                    ttsManager.speak("cargando voz inteligente",
                        onStart = { enterTtsSpeaking() },
                        onComplete = {
                            ttsManager.setEngineAsync(TtsManager.EngineMode.SHERPA) { ok, msg ->
                                mainHandler.post {
                                    log("setEngine(SHERPA): ok=$ok, msg=$msg")
                                    speakAndReturn(msg) { afterExecution() }
                                }
                            }
                        })
                } else {
                    ttsManager.setEngineAsync(TtsManager.EngineMode.ESPEAK) { ok, msg ->
                        mainHandler.post {
                            log("setEngine(ESPEAK): ok=$ok, msg=$msg")
                            speakAndReturn(msg) { afterExecution() }
                        }
                    }
                }
            }
            is CommandResult.OpenApp -> {
                val r = appLauncher.launch(parsed.appName)
                val message = when (r) {
                    is AppLauncher.Result.Ok -> "abriendo ${r.appName}"
                    is AppLauncher.Result.NotFound -> "no encuentro la app ${parsed.appName}"
                    is AppLauncher.Result.Error -> "no pude abrir ${parsed.appName}"
                }
                log("OpenApp(\"${parsed.appName}\") -> $message")
                speakAndReturn(message) { afterExecution() }
            }
            is CommandResult.CloseApp -> {
                val pkg = appLauncher.findPackage(parsed.appName)
                if (pkg == null) {
                    speakAndReturn("no encuentro la app ${parsed.appName}") { afterExecution() }
                } else {
                    log("CloseApp(\"${parsed.appName}\") pkg=$pkg")
                    appLauncher.bringVocaliaToFront()
                    mainHandler.postDelayed({
                        val r = appLauncher.closeApp(parsed.appName)
                        val message = when (r) {
                            is AppLauncher.Result.Ok -> "cerrando ${r.appName}"
                            is AppLauncher.Result.NotFound -> "no encuentro la app ${parsed.appName}"
                            is AppLauncher.Result.Error -> "no pude cerrar ${parsed.appName}"
                        }
                        log("CloseApp(\"${parsed.appName}\") -> $message")
                        speakAndReturn(message) { afterExecution() }
                    }, 600)
                }
            }
            CommandResult.CloseForegroundApp -> {
                log("CloseForegroundApp: trayendo Vocalia al frente")
                appLauncher.bringVocaliaToFront()
                speakAndReturn("cerrando app actual") { afterExecution() }
            }

            // ============================================================
            // Modo usar
            // ============================================================
            CommandResult.EnterUiMode -> {
                val svc = VocaliaAccessibilityService.instance
                if (svc == null) {
                    log("EnterUiMode: accesibilidad no disponible")
                    speakAndReturn("servicio de accesibilidad no disponible, actívalo en ajustes") {
                        enterIdle()
                    }
                } else {
                    log("EnterUiMode: activando modo usar")
                    speakAndReturn("modo usar activado") { enterUiNavigation() }
                }
            }
            CommandResult.ExitUiMode -> {
                log("ExitUiMode")
                speakAndReturn("modo usar desactivado") { enterIdle() }
            }
            CommandResult.UiNext     -> a11yCommand { it.nextNode() }
            CommandResult.UiPrevious -> a11yCommand { it.previousNode() }
            CommandResult.UiRead     -> a11yCommand { it.readCurrentNode() }
            CommandResult.UiClick    -> a11yCommand { it.clickCurrentNode() }
            CommandResult.UiBack     -> a11yCommand { it.performBack() }
            CommandResult.UiPaste    -> a11yCommand { it.pasteClipboard() }
            is CommandResult.UiScroll -> {
                val dir = when (parsed.direction) {
                    CommandResult.UiDirection.UP    -> VocaliaAccessibilityService.ScrollDirection.UP
                    CommandResult.UiDirection.DOWN  -> VocaliaAccessibilityService.ScrollDirection.DOWN
                    CommandResult.UiDirection.LEFT  -> VocaliaAccessibilityService.ScrollDirection.LEFT
                    CommandResult.UiDirection.RIGHT -> VocaliaAccessibilityService.ScrollDirection.RIGHT
                }
                a11yCommand { it.scroll(dir) }
            }
        }
    }

    // ----------------------------------------------------------------
    // Utilidades
    // ----------------------------------------------------------------

    private fun extractWake(text: String): Pair<Boolean, String> {
        val lower = text.lowercase().trim()
        for (w in WAKE_WORDS) {
            if (lower.startsWith(w)) {
                val remainder = lower.substring(w.length).trim()
                return true to remainder
            }
        }
        return false to ""
    }

    private fun extractText(json: String): String = try {
        JSONObject(json).optString("text", "").trim()
    } catch (_: Exception) { "" }

    private fun cancelTimeouts() {
        timeoutRunnable?.let { mainHandler.removeCallbacks(it) }; timeoutRunnable = null
        safetyRunnable?.let { mainHandler.removeCallbacks(it) }; safetyRunnable = null
        resumeRunnable?.let { mainHandler.removeCallbacks(it) }; resumeRunnable = null
        uiTimeoutRunnable?.let { mainHandler.removeCallbacks(it) }; uiTimeoutRunnable = null
    }
}

        'с' to "s", 'т' to "t", 'у' to "u", 'ф' to "f", 'х' to "j", 'ц' to "ts",
        'ч' to "ch", 'ш' to "sh", 'щ' to "shch", 'ъ' to "", 'ы' to "y", 'ь' to "",
        'э' to "e", 'ю' to "yu", 'я' to "ya"
    )

    @Volatile private var cachedInstalledApps: List<Pair<String, String>>? = null

    fun preload() {
        cachedInstalledApps = loadInstalledApps()
        val apps = cachedInstalledApps ?: emptyList()
        InternalDebug.mark("AppLauncher: ${apps.size} apps instaladas")
        apps.sortedBy { it.first.lowercase() }.forEach { (label, pkg) ->
            InternalDebug.mark("AppLauncher:   - $label ($pkg) -> TTS: ${transliterateCyrillic(label)}")
        }
        InternalDebug.mark("AppLauncher: ${knownApps.size} aliases")
    }

    fun getAllInstalledAppNames(): List<String> =
        getInstalledApps().map { it.first.lowercase() }.distinct().sorted()

    fun getAllAppsForTTS(): List<String> =
        getInstalledApps().map { transliterateCyrillic(it.first) }.distinct().sorted()

    fun getKnownAliases(): Set<String> = knownApps.keys

    fun getAliasesForApp(appLabel: String): List<String> {
        val pkg = findPackage(appLabel) ?: return emptyList()
        return knownApps.filterValues { it.contains(pkg) }.keys.toList().sorted()
    }

    fun hasApp(name: String): Boolean = findPackage(name) != null

    fun findPackage(name: String): String? {
        val query = normalize(name)
        if (query.isBlank()) return null

        val curated = knownApps[query]
        if (curated != null) {
            for (pkg in curated) {
                try {
                    if (context.packageManager.getLaunchIntentForPackage(pkg) != null) return pkg
                } catch (_: Throwable) {}
            }
        }

        val apps = getInstalledApps()
        val normalizedApps = apps.map { normalize(transliterateCyrillic(it.first)) to it }
        val exact = normalizedApps.firstOrNull { it.first == query }?.second
        val startsWith = exact ?: normalizedApps.firstOrNull { it.first.startsWith(query) }?.second
        val wordMatch = startsWith ?: normalizedApps.firstOrNull { (label, _) ->
            label.split("\\s+".toRegex()).any { it == query }
        }?.second
        return wordMatch?.second
    }

    private fun getInstalledApps(): List<Pair<String, String>> =
        cachedInstalledApps ?: loadInstalledApps().also { cachedInstalledApps = it }

    private fun loadInstalledApps(): List<Pair<String, String>> {
        val pm = context.packageManager
        val intent = Intent(Intent.ACTION_MAIN, null).apply { addCategory(Intent.CATEGORY_LAUNCHER) }
        return try {
            pm.queryIntentActivities(intent, 0).mapNotNull { ri: ResolveInfo ->
                val label = ri.loadLabel(pm)?.toString() ?: return@mapNotNull null
                val pkg = ri.activityInfo?.packageName ?: return@mapNotNull null
                label to pkg
            }
        } catch (t: Throwable) {
            InternalDebug.mark("AppLauncher: loadInstalledApps THREW ${t.javaClass.simpleName}: ${t.message}")
            emptyList()
        }
    }

    fun launch(appName: String): Result {
        val query = normalize(appName)
        InternalDebug.mark("AppLauncher: launch(\"$appName\") norm=\"$query\"")

        val curated = knownApps[query]
        if (curated != null) {
            for (pkg in curated) {
                if (launchPackage(pkg)) {
                    val label = getLabelForPackage(pkg) ?: appName
                    InternalDebug.mark("AppLauncher: launch OK -> $pkg ($label)")
                    return Result.Ok(label)
                }
            }
            InternalDebug.mark("AppLauncher: curado \"$query\" sin match instalado")
        }

        val pkg = findPackage(appName)
        if (pkg != null) {
            InternalDebug.mark("AppLauncher: match \"$query\" -> $pkg")
            if (launchPackage(pkg)) {
                val label = getLabelForPackage(pkg) ?: appName
                return Result.Ok(label)
            }
        }

        InternalDebug.mark("AppLauncher: NO encontrado \"$query\"")
        return Result.NotFound(appName)
    }

    private fun launchPackage(pkg: String): Boolean = try {
        val launchIntent = context.packageManager.getLaunchIntentForPackage(pkg) ?: run {
            InternalDebug.mark("AppLauncher: sin launchIntent para $pkg")
            return false
        }
        launchIntent.addFlags(
            Intent.FLAG_ACTIVITY_NEW_TASK
                or Intent.FLAG_ACTIVITY_REORDER_TO_FRONT
                or Intent.FLAG_ACTIVITY_BROUGHT_TO_FRONT
                or Intent.FLAG_ACTIVITY_SINGLE_TOP
        )
        context.startActivity(launchIntent)
        InternalDebug.mark("AppLauncher: startActivity OK -> $pkg")
        true
    } catch (t: Throwable) {
        InternalDebug.mark("AppLauncher: launchPackage($pkg) THREW ${t.javaClass.simpleName}: ${t.message}")
        false
    }

    fun bringVocaliaToFront() {
        try {
            val intent = Intent(context, Class.forName("io.github.stringmanolo.vocalia.MainActivity"))
            intent.addFlags(
                Intent.FLAG_ACTIVITY_NEW_TASK
                    or Intent.FLAG_ACTIVITY_REORDER_TO_FRONT
                    or Intent.FLAG_ACTIVITY_SINGLE_TOP
            )
            context.startActivity(intent)
            InternalDebug.mark("AppLauncher: Vocalia traída al frente")
        } catch (t: Throwable) {
            InternalDebug.mark("AppLauncher: bringVocaliaToFront THREW ${t.javaClass.simpleName}: ${t.message}")
        }
    }

    fun closeApp(appName: String): Result {
        val pkg = findPackage(appName)
        if (pkg == null) {
            InternalDebug.mark("AppLauncher: closeApp NO encontrado \"$appName\"")
            return Result.NotFound(appName)
        }
        val am = context.getSystemService(Context.ACTIVITY_SERVICE) as ActivityManager
        val foreground = isInForeground(pkg)
        InternalDebug.mark("AppLauncher: closeApp(\"$appName\") pkg=$pkg foreground=$foreground")
        am.killBackgroundProcesses(pkg)
        InternalDebug.mark("AppLauncher: killBackgroundProcesses OK -> $pkg")
        val label = getLabelForPackage(pkg) ?: appName
        return Result.Ok(label)
    }

    private fun isInForeground(pkg: String): Boolean = try {
        @Suppress("DEPRECATION")
        val tasks = (context.getSystemService(Context.ACTIVITY_SERVICE) as ActivityManager)
            .getRunningTasks(1)
        tasks.isNotEmpty() && tasks[0].topActivity?.packageName == pkg
    } catch (_: Throwable) { false }

    /** Devuelve el label visible TRANSLITERADO si es cirílico. */
    private fun getLabelForPackage(pkg: String): String? = try {
        val pm = context.packageManager
        val appInfo = pm.getApplicationInfo(pkg, 0)
        transliterateCyrillic(pm.getApplicationLabel(appInfo).toString())
    } catch (_: Throwable) { null }

    private fun transliterateCyrillic(input: String): String {
        val sb = StringBuilder()
        for (ch in input) {
            sb.append(cyrillicToSpanish[ch] ?: ch.toString())
        }
        return sb.toString()
    }

    private fun normalize(s: String): String =
        Normalizer.normalize(s.lowercase().trim(), Normalizer.Form.NFD)
            .replace("\\p{Mn}+".toRegex(), "")
}

package io.github.stringmanolo.vocalia.tts

import android.content.Context
import io.github.stringmanolo.vocalia.util.InternalDebug
import java.util.concurrent.CountDownLatch
import java.util.concurrent.TimeUnit

/**
 * Gestor de TTS con chunking por frase.
 *
 * Cada chunk es una unidad reanudable: al pausar se guarda desde el chunk actual en adelante
 * en `pendingChunks`. Al reanudar, se re-lee el chunk actual desde su inicio (la frase completa).
 * Esto permite en el futuro: repetir última frase, saltar N frases, etc.
 */
class TtsManager(private val context: Context) {

    enum class EngineMode { ESPEAK, SHERPA }

    var onEngineReady: ((TtsEngine) -> Unit)? = null

    @Volatile private var currentMode: EngineMode = EngineMode.ESPEAK
    @Volatile private var currentEngine: TtsEngine? = null

    private var espeakEngine: EspeakNgEngine? = null
    private var sherpaEngine: SherpaOnnxEngine? = null
    @Volatile private var sherpaReady = false
    @Volatile private var sherpaLoading = false

    @Volatile private var speaking = false
    @Volatile private var paused = false
    @Volatile private var stopped = false
    @Volatile private var speakThread: Thread? = null
    @Volatile private var currentChunkText: String? = null

    /** Lista completa de chunks de la lectura actual (para navegación futura). */
    @Volatile private var chunks: List<String> = emptyList()
    /** Índice del chunk que se está reproduciendo (o se iba a reproducir). */
    @Volatile private var currentChunkIndex: Int = -1
    /** Chunks pendientes cuando se pausó (incluye el chunk en curso). */
    @Volatile private var pendingChunks: List<String> = emptyList()
    /** Índice del primer chunk pendiente dentro de `chunks`. */
    @Volatile private var pendingStartIndex: Int = -1

    private fun dbg(msg: String) { InternalDebug.mark("[${Thread.currentThread().name}] TtsManager: $msg") }

    fun init() {
        val espeak = EspeakNgEngine(context)
        espeakEngine = espeak
        espeak.init(
            onReady = {
                dbg("EspeakNg listo")
                currentEngine = espeak
                currentMode = EngineMode.ESPEAK
                onEngineReady?.invoke(espeak)
            },
            onError = { dbg("EspeakNg falló: $it") }
        )
    }

    fun setEngineAsync(mode: EngineMode, callback: (Boolean, String) -> Unit) {
        when (mode) {
            EngineMode.ESPEAK -> {
                val e = espeakEngine
                if (e != null) {
                    currentEngine = e
                    currentMode = EngineMode.ESPEAK
                    callback(true, "voz por defecto activada")
                } else callback(false, "espeak no disponible")
            }
            EngineMode.SHERPA -> {
                if (sherpaReady) {
                    currentEngine = sherpaEngine
                    currentMode = EngineMode.SHERPA
                    callback(true, "voz inteligente activada")
                    return
                }
                if (sherpaLoading) { callback(false, "voz inteligente cargándose"); return }
                sherpaLoading = true
                Thread({
                    try {
                        val sherpa = SherpaOnnxEngine(context)
                        sherpa.init(
                            onReady = {
                                sherpaEngine = sherpa
                                sherpaReady = true
                                sherpaLoading = false
                                currentEngine = sherpa
                                currentMode = EngineMode.SHERPA
                                callback(true, "voz inteligente activada")
                            },
                            onError = {
                                sherpaEngine = null
                                sherpaReady = false
                                sherpaLoading = false
                                callback(false, "voz inteligente no disponible")
                            }
                        )
                    } catch (t: Throwable) {
                        sherpaEngine = null
                        sherpaReady = false
                        sherpaLoading = false
                        callback(false, "voz inteligente no disponible")
                    }
                }, "sherpa-load").start()
            }
        }
    }

    /**
     * Habla [text] troceado por frases. Cada frase es un chunk reanudable.
     */
    fun speak(text: String, onStart: (() -> Unit)? = null, onComplete: (() -> Unit)? = null) {
        if (speaking) {
            dbg("speak: ya hablando, ignorando")
            onComplete?.invoke()
            return
        }
        val engine = currentEngine ?: run { onComplete?.invoke(); return }

        speaking = true
        paused = false
        stopped = false
        pendingChunks = emptyList()
        pendingStartIndex = -1

        chunks = splitIntoChunks(text)
        currentChunkIndex = 0
        dbg("speak: ${chunks.size} chunks, ${text.length} chars")

        launchSpeakThread(engine, startIndex = 0, onStart, onComplete)
    }

    /**
     * Reanuda la lectura pausada desde el chunk pendiente.
     * Devuelve true si había algo que reanudar.
     */
    fun resumePaused(onStart: (() -> Unit)? = null, onComplete: (() -> Unit)? = null): Boolean {
        val pending = pendingChunks
        if (pending.isEmpty() || pendingStartIndex < 0) {
            dbg("resumePaused: no hay chunks pendientes")
            return false
        }
        if (speaking) {
            dbg("resumePaused: ya hablando, ignorando")
            return false
        }
        val engine = currentEngine ?: return false

        dbg("resumePaused: desde chunk $pendingStartIndex, ${pending.size} pendientes")
        val start = pendingStartIndex
        pendingChunks = emptyList()
        pendingStartIndex = -1
        speaking = true
        paused = false
        stopped = false
        currentChunkIndex = start
        launchSpeakThread(engine, startIndex = start, onStart, onComplete)
        return true
    }

    private fun launchSpeakThread(
        engine: TtsEngine,
        startIndex: Int,
        onStart: (() -> Unit)?,
        onComplete: (() -> Unit)?
    ) {
        val localChunks = chunks
        val th = Thread({
            var started = false
            try {
                var idx = startIndex
                while (idx < localChunks.size) {
                    if (stopped) { dbg("speak: stop en chunk $idx"); break }

                    if (paused) {
                        engine.pause()
                        currentChunkIndex = idx
                        pendingChunks = localChunks.subList(idx, localChunks.size)
                        pendingStartIndex = idx
                        dbg("speak: pausado antes de chunk $idx (pendientes=${pendingChunks.size})")
                        while (paused && !stopped) Thread.sleep(100)
                        if (stopped) break
                        engine.resume()
                        pendingChunks = emptyList()
                        pendingStartIndex = -1
                        dbg("speak: reanudado en chunk $idx")
                    }

                    currentChunkIndex = idx
                    val chunk = localChunks[idx]
                    currentChunkText = chunk
                    if (!started) { started = true; onStart?.invoke() }

                    val latch = CountDownLatch(1)
                    engine.onComplete = { latch.countDown() }
                    engine.speak(chunk)
                    latch.await(180, TimeUnit.SECONDS)

                    // ¿Se pausó durante este chunk? Guardar desde aquí (incluido él) y esperar.
                    if (paused && !stopped) {
                        pendingChunks = localChunks.subList(idx, localChunks.size)
                        pendingStartIndex = idx
                        dbg("speak: pausado durante chunk $idx (pendientes=${pendingChunks.size})")
                        while (paused && !stopped) Thread.sleep(100)
                        if (stopped) break
                        engine.resume()
                        pendingChunks = emptyList()
                        pendingStartIndex = -1
                        dbg("speak: reanudado tras chunk $idx (re-lee)")
                        // Re-lee este mismo chunk
                        continue
                    }

                    idx++
                }
            } catch (t: Throwable) {
                dbg("speak: THREW ${t.javaClass.simpleName}: ${t.message}")
            } finally {
                currentChunkText = null
                speaking = false
                paused = false
                stopped = false
                pendingChunks = emptyList()
                pendingStartIndex = -1
                onComplete?.invoke()
            }
        }, "tts-chunk")
        speakThread = th
        th.start()
    }

    fun pause() {
        dbg("pause")
        paused = true
        currentEngine?.pause()
    }

    fun resume() {
        dbg("resume")
        paused = false
        currentEngine?.resume()
    }

    fun isSpeaking(): Boolean = speaking
    fun isPaused(): Boolean = paused
    fun getCurrentChunkText(): String? = currentChunkText

    /** ¿Hay chunks pendientes de reanudar? */
    fun hasPausedSpeech(): Boolean = pendingChunks.isNotEmpty()

    /** Descarta los chunks pendientes (al abandonar la lectura pausada). */
    fun clearPausedSpeech() {
        pendingChunks = emptyList()
        pendingStartIndex = -1
    }

    // --- API preparada para el futuro (navegación por frases) ---

    fun getChunks(): List<String> = chunks
    fun getCurrentChunkIndex(): Int = currentChunkIndex
    fun getCurrentChunk(): String? = currentChunkText

    fun stop() {
        dbg("stop")
        stopped = true
        paused = false
        pendingChunks = emptyList()
        pendingStartIndex = -1
        chunks = emptyList()
        currentChunkIndex = -1
        currentEngine?.stop()
        try { speakThread?.join(2000) } catch (_: InterruptedException) {}
        currentChunkText = null
    }

    fun shutdown() {
        dbg("shutdown")
        stopped = true
        pendingChunks = emptyList()
        pendingStartIndex = -1
        try { speakThread?.join(2000) } catch (_: InterruptedException) {}
        try { espeakEngine?.shutdown() } catch (_: Throwable) {}
        try { sherpaEngine?.shutdown() } catch (_: Throwable) {}
        currentEngine = null
        espeakEngine = null
        sherpaEngine = null
        sherpaReady = false
        sherpaLoading = false
    }

    /**
     * Trocea el texto en frases. Una frase puede contener varias sub-frases si el bloque
     * es muy corto (para no generar pausas antinaturales), y se corta en el último delimitador
     * fuerte si excede [MAX_CHUNK_CHARS].
     */
    private fun splitIntoChunks(text: String): List<String> {
        val clean = text.trim()
        if (clean.isEmpty()) return emptyList()

        // 1) Frases terminadas en . ! ? (se comen los espacios)
        val sentences = Regex("[^.!?]+[.!?]+").findAll(clean)
            .map { it.value.trim() }
            .filter { it.isNotEmpty() }
            .toList()

        val base: List<String> = if (sentences.isEmpty()) listOf(clean) else sentences

        // 2) Une frases muy cortas y parte frases muy largas
        val out = mutableListOf<String>()
        val buf = StringBuilder()

        fun flush() {
            if (buf.isNotEmpty()) {
                out.add(buf.toString().trim())
                buf.clear()
            }
        }

        for (s in base) {
            if (s.length > MAX_CHUNK_CHARS) {
                flush()
                // Parte por , ; : o por espacio, en trozos <= MAX_CHUNK_CHARS
                var rest = s
                while (rest.length > MAX_CHUNK_CHARS) {
                    val cut = lastSplit(rest, MAX_CHUNK_CHARS)
                    out.add(rest.substring(0, cut).trim())
                    rest = rest.substring(cut).trim()
                }
                if (rest.isNotEmpty()) out.add(rest)
                continue
            }
            if (buf.length + s.length + 1 <= MAX_CHUNK_CHARS) {
                if (buf.isNotEmpty()) buf.append(' ')
                buf.append(s)
            } else {
                flush()
                buf.append(s)
            }
        }
        flush()
        return out.ifEmpty { listOf(clean) }
    }

    private fun lastSplit(s: String, limit: Int): Int {
        val window = s.substring(0, limit)
        val cut = listOf(',', ';', ':', ' ').map { window.lastIndexOf(it) }.maxOrNull() ?: -1
        return if (cut > 0) cut + 1 else limit
    }

    companion object {
        /** Tamaño máximo de un chunk. Si se supera, se parte por , ; : o espacio. */
        private const val MAX_CHUNK_CHARS = 240
    }
}
